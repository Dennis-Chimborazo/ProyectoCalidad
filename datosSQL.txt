    v_subtotal_bruto NUMERIC(12,2);
    v_descuento NUMERIC(12,2);
    v_subtotal_final NUMERIC(12,2);
    v_iva NUMERIC(12,2);
    v_total NUMERIC(12,2);

    v_referencia TEXT;
    v_fecha TIMESTAMP;
    v_tiempo_respuesta NUMERIC(10,2);

    v_productos_usados INTEGER[];
BEGIN
    WHILE v_contador_ventas <= v_num_ventas LOOP

        -- Seleccionar cliente activo aleatorio
        SELECT id, name, cedula
        INTO v_cliente
        FROM techstore_cliente
        WHERE activo = true
        ORDER BY random()
        LIMIT 1;

        IF v_cliente.id IS NULL THEN
            RAISE EXCEPTION 'No existen clientes activos para generar ventas.';
        END IF;

        -- Crear referencia y valores iniciales
        v_referencia := 'VTA-DEMO-' || to_char(NOW(), 'YYYYMMDDHH24MISS') || '-' || LPAD(v_contador_ventas::TEXT, 4, '0');
        v_fecha := NOW() - ((random() * 30)::INTEGER || ' days')::INTERVAL;
        v_tiempo_respuesta := ROUND((0.10 + random() * 2.40)::NUMERIC, 2);

        v_subtotal_bruto := 0;
        v_descuento := 0;
        v_subtotal_final := 0;
        v_iva := 0;
        v_total := 0;
        v_productos_usados := ARRAY[]::INTEGER[];

        -- Crear cabecera de venta temporalmente con totales en 0
        INSERT INTO techstore_venta
        (
            id,
            name,
            cliente_id,
            subtotal,
            descuento,
            iva,
            total,
            tiempo_respuesta,
            estado_calidad,
            observacion,
            create_uid,
            create_date,
            write_uid,
            write_date
        )
        VALUES
        (
            nextval('techstore_venta_id_seq'),
            v_referencia,
            v_cliente.id,
            0,
            0,
            0,
            0,
            v_tiempo_respuesta,
            'aceptable',
            'Venta generada automáticamente para pruebas con detalles aleatorios.',
            1,
            v_fecha,
            1,
            v_fecha
        )
        RETURNING id INTO v_venta_id;

        -- Cada venta tendrá entre 1 y 10 productos
        v_num_detalles := 1 + FLOOR(random() * 10)::INTEGER;
        v_contador_detalles := 1;

        WHILE v_contador_detalles <= v_num_detalles LOOP

            -- Seleccionar producto activo con stock, evitando repetir producto en la misma venta
            SELECT id, name, codigo, precio_unitario, stock_disponible
            INTO v_producto
            FROM techstore_producto
            WHERE activo = true
              AND stock_disponible > 0
              AND NOT (id = ANY(v_productos_usados))
            ORDER BY random()
            LIMIT 1;

            -- Si ya no hay productos disponibles, salir del detalle
            IF v_producto.id IS NULL THEN
                EXIT;
            END IF;

            -- Cantidad aleatoria segura: de 1 a 3, sin superar stock
            v_cantidad := 1 + FLOOR(random() * 3)::INTEGER;

            IF v_cantidad > v_producto.stock_disponible THEN
                v_cantidad := v_producto.stock_disponible;
            END IF;

            IF v_cantidad <= 0 THEN
                EXIT;
            END IF;

            v_subtotal_linea := ROUND((v_cantidad * v_producto.precio_unitario)::NUMERIC, 2);

            -- Insertar detalle
            INSERT INTO techstore_detalle_venta
            (
                id,
                venta_id,
                producto_id,
                cantidad,
                precio_unitario,
                subtotal,
                create_uid,
                create_date,
                write_uid,
                write_date
            )
            VALUES
            (
                nextval('techstore_detalle_venta_id_seq'),
                v_venta_id,
                v_producto.id,
                v_cantidad,
                v_producto.precio_unitario,
                v_subtotal_linea,
                1,
                v_fecha,
                1,
                v_fecha
            );

            -- Descontar stock
            UPDATE techstore_producto
            SET stock_disponible = stock_disponible - v_cantidad,
                write_uid = 1,
                write_date = v_fecha
            WHERE id = v_producto.id;

            -- Acumular subtotal
            v_subtotal_bruto := v_subtotal_bruto + v_subtotal_linea;

            -- Guardar producto usado para no repetirlo en la misma venta
            v_productos_usados := array_append(v_productos_usados, v_producto.id);

            v_contador_detalles := v_contador_detalles + 1;
        END LOOP;

        -- Si por alguna razón no se agregaron detalles, eliminar venta
        IF v_subtotal_bruto <= 0 THEN
            DELETE FROM techstore_venta WHERE id = v_venta_id;
        ELSE
            -- Descuento aleatorio entre 0% y 10% del subtotal
            v_descuento := ROUND((v_subtotal_bruto * (random() * 0.10))::NUMERIC, 2);

            v_subtotal_final := v_subtotal_bruto - v_descuento;
            v_iva := ROUND((v_subtotal_final * 0.12)::NUMERIC, 2);
            v_total := ROUND((v_subtotal_final + v_iva)::NUMERIC, 2);

            -- Actualizar totales de la venta
            UPDATE techstore_venta
            SET subtotal = v_subtotal_final,
                descuento = v_descuento,
                iva = v_iva,
                total = v_total,
                write_uid = 1,
                write_date = v_fecha
            WHERE id = v_venta_id;
        END IF;

        v_contador_ventas := v_contador_ventas + 1;
    END LOOP;
END $$;

            )
            RETURNING id INTO v_mantenimiento_equipo_id;

            INSERT INTO techstore_mantenimiento_equipo_falla
            (
                mantenimiento_equipo_id,
                base_conocimiento_id,
                create_uid,
                write_uid,
                create_date,
                write_date
            )
            VALUES
            (
                v_mantenimiento_equipo_id,
                v_base.id,
                1,
                1,
                v_fecha_inicio,
                v_fecha_inicio
            );

            v_total_equipos := v_total_equipos + 1;
            v_contador_equipo := v_contador_equipo + 1;
        END LOOP;

        -- Actualizar total_equipos porque el campo es compute store y por SQL directo no se recalcula solo.
        UPDATE techstore_mantenimiento
        SET total_equipos = v_total_equipos,
            write_uid = 1,
            write_date = COALESCE(v_fecha_fin, v_fecha_inicio)
        WHERE id = v_mantenimiento_id;

        -- Si por algún caso no se insertó ningún equipo, eliminar mantenimiento vacío.
        IF v_total_equipos = 0 THEN
            DELETE FROM techstore_mantenimiento
            WHERE id = v_mantenimiento_id;
        END IF;

        v_contador := v_contador + 1;
    END LOOP;
END $$;

SELECT COUNT(*) AS total_mantenimientos
FROM techstore_mantenimiento;

SELECT estado, COUNT(*) AS cantidad
FROM techstore_mantenimiento
GROUP BY estado
ORDER BY estado;

SELECT prioridad, COUNT(*) AS cantidad
FROM techstore_mantenimiento
GROUP BY prioridad
ORDER BY prioridad;

SELECT COUNT(*) AS total_equipos
FROM techstore_equipo;

SELECT COUNT(*) AS total_detalles_equipo
FROM techstore_detalle_equipo;

SELECT estado_equipo, COUNT(*) AS cantidad
FROM techstore_mantenimiento_equipo
GROUP BY estado_equipo
ORDER BY estado_equipo;

SELECT COUNT(*) AS total_fallas_asociadas
FROM techstore_mantenimiento_equipo_falla;

SELECT
    m.id,
    m.name,
    c.name AS cliente,
    t.name AS tecnico,
    m.estado,
    m.prioridad,
    m.total_equipos,
    m.fecha_inicio,
    m.fecha_fin
FROM techstore_mantenimiento m
JOIN techstore_cliente c ON c.id = m.cliente_id
JOIN techstore_tecnico t ON t.id = m.tecnico_id
ORDER BY m.id DESC
LIMIT 10;
